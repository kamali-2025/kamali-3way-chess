    const RADIUS = 5;
    const HEX_SIZE = 31;
    const SEATS = [

    const PIECE_VALUES = { P: 1, N: 3, B: 3, R: 5, Q: 9, K: 15 };

      K: "♔", Q: "♕", R: "♖", B: "♗", N: "♘", P: "♙"
    };

    // Axial directions (6 orthogonal neighbors on hex grid)
    const ORTHO_DIRS = [
      [1, 0], [1, -1], [0, -1],
      [-1, 0], [-1, 1], [0, 1]
    ];
    // 6 diagonal directions on hex grid
    const DIAG_DIRS = [
      [2, -1], [1, -2], [-1, -1],
      [-2, 1], [-1, 2], [1, 1]
    ];
    // 12 Knight jumps on hex grid
    const KNIGHT_DIRS = [
      [3, -1], [3, -2], [2, -3], [1, -3],
      [-1, -2], [-2, -1], [-3, 1], [-3, 2],
      [-2, 3], [-1, 3], [1, 2], [2, 1]

    function keyOf(q, r) { return `${q},${r}`; }
    function parseKey(k) { return k.split(",").map(Number); }
    function isValidHex(q, r) {
      const s = -q - r;
      return Math.max(Math.abs(q), Math.abs(r), Math.abs(s)) <= RADIUS;
    }

    function formatCoord(q, r) {
      const file = String.fromCharCode(65 + (q + RADIUS)); // A..K
      const rank = r + RADIUS + 1; // 1..11
      return `${file}${rank}`;
    }

    // Simple Web Audio synth
    function playTone(type) {
      if (!soundOn) return;
      try {
        const ctx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.connect(gain);
        gain.connect(ctx.destination);
        const now = ctx.currentTime;
        if (type === "move") {
          osc.frequency.setValueAtTime(340, now);
          osc.frequency.exponentialRampToValueAtTime(520, now + 0.08);
          gain.gain.setValueAtTime(0.08, now);
        } else if (type === "capture") {
          osc.type = "triangle";
          osc.frequency.setValueAtTime(240, now);
          osc.frequency.exponentialRampToValueAtTime(110, now + 0.16);
          gain.gain.setValueAtTime(0.14, now);
        } else if (type === "check") {
          osc.type = "sawtooth";
          osc.frequency.setValueAtTime(620, now);
          osc.frequency.setValueAtTime(840, now + 0.09);
          gain.gain.setValueAtTime(0.11, now);
        }
        osc.start(now);
        osc.stop(now + 0.18);
      } catch (e) {}
    }

    function setupInitialPieces() {
      board.clear();
      // White (Seat 0 - Bottom, moves toward r = -1)
      const whiteBack = [
        [-4, 5, "R"], [-3, 5, "N"], [-2, 5, "B"], [-1, 5, "Q"],
        [0, 5, "K"], [1, 4, "B"], [2, 3, "N"], [3, 2, "R"]
      ];
      whiteBack.forEach(([q, r, t]) => board.set(keyOf(q, r), { seat: 0, type: t }));
      for (let q = -4; q <= 4; q++) {
        const r = q <= 0 ? 4 : 4 - q;
        if (isValidHex(q, r)) board.set(keyOf(q, r), { seat: 0, type: "P" });
      }

      // Crimson (Seat 1 - Top-Left, moves toward +q, +r)
      const crimsonBack = [
        [-5, 0, "R"], [-5, 1, "N"], [-5, 2, "B"], [-5, 3, "Q"],
        [-5, 4, "K"], [-4, -1, "B"], [-3, -2, "N"], [-2, -3, "R"]
      ];
      crimsonBack.forEach(([q, r, t]) => board.set(keyOf(q, r), { seat: 1, type: t }));
      for (let i = -4; i <= 4; i++) {
        const q = i < 0 ? -4 : -4 + i;
        const r = i < 0 ? Math.abs(i) - 1 : -4;
        if (isValidHex(q, r) && !board.has(keyOf(q, r))) {
          board.set(keyOf(q, r), { seat: 1, type: "P" });
        }
      }

      // Emerald (Seat 2 - Top-Right, moves toward -q, +r)
      const emeraldBack = [
        [2, -5, "R"], [3, -5, "N"], [4, -5, "B"], [5, -5, "Q"],
        [5, -4, "K"], [5, -3, "B"], [5, -2, "N"], [5, -1, "R"]
      ];
      emeraldBack.forEach(([q, r, t]) => board.set(keyOf(q, r), { seat: 2, type: t }));
      for (let i = 0; i <= 7; i++) {
        const q = i < 4 ? 1 + i : 4;
        const r = i < 4 ? -4 : -4 + (i - 3);
        if (isValidHex(q, r) && !board.has(keyOf(q, r))) {
          board.set(keyOf(q, r), { seat: 2, type: "P" });
        }
      }
    }

    function getPseudoMoves(fromKey, customBoard = board) {
      const piece = customBoard.get(fromKey);
      if (!piece) return [];
      const [q, r] = parseKey(fromKey);
      const moves = [];

      const addSlide = (dirs, maxSteps = 12) => {
        for (const [dq, dr] of dirs) {
          for (let step = 1; step <= maxSteps; step++) {
            const nq = q + dq * step;
            const nr = r + dr * step;
            if (!isValidHex(nq, nr)) break;
            const tk = keyOf(nq, nr);
            const occ = customBoard.get(tk);
            if (!occ) {
              moves.push(tk);
            } else {
              if (occ.seat !== piece.seat) moves.push(tk);
              break;
            }
          }
        }
      };

      if (piece.type === "R") addSlide(ORTHO_DIRS);
      else if (piece.type === "B") addSlide(DIAG_DIRS);
      else if (piece.type === "Q") addSlide([...ORTHO_DIRS, ...DIAG_DIRS]);
      else if (piece.type === "K") addSlide([...ORTHO_DIRS, ...DIAG_DIRS], 1);
      else if (piece.type === "N") {
        for (const [dq, dr] of KNIGHT_DIRS) {
          const nq = q + dq, nr = r + dr;
          if (!isValidHex(nq, nr)) continue;
          const tk = keyOf(nq, nr);
          const occ = customBoard.get(tk);
          if (!occ || occ.seat !== piece.seat) moves.push(tk);
        }
      } else if (piece.type === "P") {
        // Forward & capture vectors per seat
        const forwardDirs = piece.seat === 0 ? [[0, -1], [1, -1]] :
                            piece.seat === 1 ? [[1, 0], [1, -1]] :
                                               [[-1, 1], [-1, 0]];
        const capDirs = piece.seat === 0 ? [[-1, 0], [1, -2], [2, -1]] :
                        piece.seat === 1 ? [[0, 1], [2, -1], [1, -2]] :
                                           [[0, -1], [-2, 1], [-1, 2]];
        for (const [dq, dr] of forwardDirs) {
          const nq = q + dq, nr = r + dr;
          if (isValidHex(nq, nr)) {
            const tk = keyOf(nq, nr);
            if (!customBoard.has(tk)) moves.push(tk);
          }
        }
        for (const [dq, dr] of capDirs) {
          const nq = q + dq, nr = r + dr;
          if (isValidHex(nq, nr)) {
            const tk = keyOf(nq, nr);
            const occ = customBoard.get(tk);
            if (occ && occ.seat !== piece.seat) moves.push(tk);
          }
        }
      }
      return moves;
    }

    function isKingInCheck(seat, customBoard = board) {
      if (eliminated[seat]) return false;
      let kingKey = null;
      for (const [k, p] of customBoard.entries()) {
        if (p.seat === seat && p.type === "K") { kingKey = k; break; }
      }
      if (!kingKey) return true;
      for (const [k, p] of customBoard.entries()) {
        if (p.seat !== seat && !eliminated[p.seat]) {
          const targets = getPseudoMoves(k, customBoard);
          if (targets.includes(kingKey)) return true;
        }
      }
      return false;
    }

    function getLegalMoves(fromKey) {
      const piece = board.get(fromKey);
      if (!piece) return [];
      const pseudo = getPseudoMoves(fromKey, board);
      return pseudo.filter(toKey => {
        const clone = new Map(board);
        clone.set(toKey, piece);
        clone.delete(fromKey);
        return !isKingInCheck(piece.seat, clone);
      });
    }

    function hasAnyLegalMoves(seat) {
      if (eliminated[seat]) return false;
      for (const [k, p] of board.entries()) {
        if (p.seat === seat && getLegalMoves(k).length > 0) return true;
      }
      return false;
    }

    function cloneState() {
      return {
        board: new Map(board),
        currentTurn,
        moveHistory: JSON.parse(JSON.stringify(moveHistory)),
        scores: [...scores],
        capturedBy: JSON.parse(JSON.stringify(capturedBy)),
        eliminated: [...eliminated],
        gameOver,
        lastMove: lastMove ? { ...lastMove } : null
      };
    }

    function restoreState(snap) {
      board = new Map(snap.board);
      currentTurn = snap.currentTurn;
      moveHistory = snap.moveHistory;
      scores = snap.scores;
      capturedBy = snap.capturedBy;
      eliminated = snap.eliminated;
      gameOver = snap.gameOver;
      lastMove = snap.lastMove;
      selectedKey = null;
      legalTargets = [];
      renderAll();
    }

    function executeMove(fromKey, toKey) {
      if (gameOver) return;
      const piece = board.get(fromKey);
      if (!piece || piece.seat !== currentTurn) return;

      undoStack.push(cloneState());

      const targetPiece = board.get(toKey);
      const [fq, fr] = parseKey(fromKey);
      const [tq, tr] = parseKey(toKey);

      board.set(toKey, piece);
      board.delete(fromKey);

      // Pawn promotion on far edge
      let promoted = false;
      if (piece.type === "P") {
        const s = -tq - tr;
        if ((piece.seat === 0 && (tr === -RADIUS || s === RADIUS)) ||
            (piece.seat === 1 && (tq === RADIUS || s === -RADIUS)) ||
            (piece.seat === 2 && (tq === -RADIUS || tr === RADIUS))) {
          board.set(toKey, { seat: piece.seat, type: "Q" });
          promoted = true;
        }
      }

      let gained = 0;
      if (targetPiece) {
        gained = PIECE_VALUES[targetPiece.type] || 1;
        scores[piece.seat] += gained;
        capturedBy[piece.seat].push(GLYPHS[targetPiece.type]);
        if (targetPiece.type === "K") {
          eliminated[targetPiece.seat] = true;
        }
      }

      lastMove = { from: fromKey, to: toKey };
      selectedKey = null;
      legalTargets = [];

      // Check opponents for check/checkmate
      let checkedOpponent = false;
      let matedSeat = -1;
      for (let s = 0; s < 3; s++) {
        if (s !== piece.seat && !eliminated[s]) {
          if (isKingInCheck(s, board)) {
            checkedOpponent = true;
            if (!hasAnyLegalMoves(s)) {
              matedSeat = s;
            }
          }
        }
      }

      const notation = `${GLYPHS[piece.type]}${formatCoord(fq, fr)}${targetPiece ? "×" : "→"}${formatCoord(tq, tr)}${promoted ? "=Q" : ""}${matedSeat !== -1 ? "#" : checkedOpponent ? "+" : ""}`;

      moveHistory.push({
        seat: piece.seat,
        notation,
        isCapture: Boolean(targetPiece),
        isCheck: checkedOpponent || matedSeat !== -1,
        gained
      });

      const mode = document.getElementById("modeSelect").value;
      const alertEl = document.getElementById("alertText");

      if (matedSeat !== -1) {
        playTone("check");
        if (mode === "firstmate") {
          gameOver = true;
          alertEl.innerHTML = `<span style="color:#facc15">🏆 CHECKMATE! ${SEATS[piece.seat].name} wins by First Mate!</span>`;
          renderAll();
          return;
        } else {
          eliminated[matedSeat] = true;
          scores[piece.seat] += 15;
          alertEl.innerHTML = `<span style="color:#fb7185">💀 ${SEATS[matedSeat].name} checkmated & eliminated (+15 pts)!</span>`;
        }
      } else if (checkedOpponent) {
        playTone("check");
        alertEl.innerHTML = `<span style="color:#f87171">⚠️ CHECK delivered by ${SEATS[piece.seat].name}!</span>`;
      } else if (targetPiece) {
        playTone("capture");
        alertEl.textContent = `${SEATS[piece.seat].name} captured ${GLYPHS[targetPiece.type]} (+${gained} pts)`;
      } else {
        playTone("move");
        alertEl.textContent = `Last: ${SEATS[piece.seat].name} played ${notation}`;
      }

      // Advance to next active seat
      advanceTurn();
      renderAll();
      scheduleAiIfNeeded();
    }

    function advanceTurn() {
      const alive = [0, 1, 2].filter(s => !eliminated[s]);
      if (alive.length <= 1) {
        gameOver = true;
        const winner = alive.length === 1 ? SEATS[alive[0]].name : "No one";
        document.getElementById("alertText").innerHTML = `<span style="color:#facc15">🏆 Match Over! ${winner} is the last standing!</span>`;
        return;
      }
      for (let i = 1; i <= 3; i++) {
        const next = (currentTurn + i) % 3;
        if (!eliminated[next]) {
          currentTurn = next;
          if (!hasAnyLegalMoves(currentTurn)) {
            eliminated[currentTurn] = true;
            advanceTurn();
          }
          return;
        }
      }
    }

    function scheduleAiIfNeeded() {
      clearTimeout(aiTimer);
      if (gameOver) return;
      if (controllers[currentTurn] === "ai") {
        aiTimer = setTimeout(runTacticalAiMove, 420);
      }
    }

    function runTacticalAiMove() {
      if (gameOver || controllers[currentTurn] !== "ai") return;
      const candidates = [];
      for (const [k, p] of board.entries()) {
        if (p.seat === currentTurn) {
          for (const tk of getLegalMoves(k)) {
            let score = Math.random() * 1.5;
            const victim = board.get(tk);
            if (victim) score += (PIECE_VALUES[victim.type] || 1) * 10;

            const clone = new Map(board);
            clone.set(tk, p);
            clone.delete(k);
            for (let s = 0; s < 3; s++) {
              if (s !== currentTurn && !eliminated[s] && isKingInCheck(s, clone)) {
                score += 18;
              }
            }
            const [tq, tr] = parseKey(tk);
            const centerDist = Math.max(Math.abs(tq), Math.abs(tr), Math.abs(-tq - tr));
            score += (RADIUS - centerDist) * 0.4;
            candidates.push({ from: k, to: tk, score });
          }
        }
      }
      if (candidates.length === 0) {
        eliminated[currentTurn] = true;
        advanceTurn();
        renderAll();
        scheduleAiIfNeeded();
        return;
      }
      candidates.sort((a, b) => b.score - a.score);
      executeMove(candidates[0].from, candidates[0].to);
    }

    function hexToPixel(q, r) {
      const x = HEX_SIZE * (Math.sqrt(3) * q + (Math.sqrt(3) / 2) * r);
      const y = HEX_SIZE * ((3 / 2) * r);
      return [x, y];
    }

    function hexPolygonPoints(cx, cy, size) {
      const pts = [];
      for (let i = 0; i < 6; i++) {
        const angle = (Math.PI / 180) * (60 * i - 30);
        pts.push(`${(cx + size * Math.cos(angle)).toFixed(1)},${(cy + size * Math.sin(angle)).toFixed(1)}`);
      }
      return pts.join(" ");
    }

    function renderBoard() {
      const svg = document.getElementById("hexBoard");
      svg.innerHTML = "";
      const tileColors = ["#273549", "#33445c", "#1e293b"];

      const checkedKings = new Set();
      for (let s = 0; s < 3; s++) {
        if (!eliminated[s] && isKingInCheck(s)) {
          for (const [k, p] of board.entries()) {
            if (p.seat === s && p.type === "K") checkedKings.add(k);
          }
        }
      }

      for (let q = -RADIUS; q <= RADIUS; q++) {
        for (let r = -RADIUS; r <= RADIUS; r++) {
          if (!isValidHex(q, r)) continue;
          const k = keyOf(q, r);
          const [cx, cy] = hexToPixel(q, r);
          const colorIdx = ((q - r) % 3 + 3) % 3;
          let fill = tileColors[colorIdx];
          let stroke = "#0f172a";
          let strokeWidth = 1.4;

          if (lastMove && (lastMove.from === k || lastMove.to === k)) {
            fill = "#4c4622";
          }
          if (selectedKey === k) {
            fill = "#0369a1";
            stroke = "#38bdf8";
            strokeWidth = 2.4;
          }
          if (checkedKings.has(k)) {
            fill = "#991b1b";
            stroke = "#ef4444";
            strokeWidth = 2.5;
          }

          const g = document.createElementNS("http://www.w3.org/2000/svg", "g");
          g.setAttribute("class", "hex-cell");
          g.addEventListener("click", () => handleHexClick(k));

          const poly = document.createElementNS("http://www.w3.org/2000/svg", "polygon");
          poly.setAttribute("points", hexPolygonPoints(cx, cy, HEX_SIZE - 0.8));
          poly.setAttribute("fill", fill);
          poly.setAttribute("stroke", stroke);
          poly.setAttribute("stroke-width", strokeWidth);
          g.appendChild(poly);

          const label = document.createElementNS("http://www.w3.org/2000/svg", "text");
          label.setAttribute("x", cx);
          label.setAttribute("y", cy + HEX_SIZE * 0.62);
          label.setAttribute("class", "coord-label");
          label.textContent = formatCoord(q, r);
          g.appendChild(label);

          const piece = board.get(k);
          if (piece) {
            const seatMeta = SEATS[piece.seat];
            const txt = document.createElementNS("http://www.w3.org/2000/svg", "text");
            txt.setAttribute("x", cx);
            txt.setAttribute("y", cy - 2);
            txt.setAttribute("class", "piece-text");
            txt.setAttribute("fill", seatMeta.color);
            txt.setAttribute("stroke", seatMeta.stroke);
            txt.setAttribute("stroke-width", "1.1");
            txt.setAttribute("paint-order", "stroke");
            txt.textContent = GLYPHS[piece.type];
            g.appendChild(txt);
          }

          if (legalTargets.includes(k)) {
            const dot = document.createElementNS("http://www.w3.org/2000/svg", "circle");
            dot.setAttribute("cx", cx);
            dot.setAttribute("cy", cy);
            dot.setAttribute("r", piece ? HEX_SIZE * 0.68 : 6.5);
            dot.setAttribute("fill", piece ? "none" : "rgba(34, 197, 94, 0.85)");
            if (piece) {
              dot.setAttribute("stroke", "#22c55e");
              dot.setAttribute("stroke-width", "3");
            }
            dot.style.pointerEvents = "none";
            g.appendChild(dot);
          }

          svg.appendChild(g);
        }
      }
    }

    function handleHexClick(k) {
      if (gameOver || controllers[currentTurn] !== "human") return;
      if (selectedKey && legalTargets.includes(k)) {
        executeMove(selectedKey, k);
        return;
      }
      const piece = board.get(k);
      if (piece && piece.seat === currentTurn) {
        selectedKey = k;
        legalTargets = getLegalMoves(k);
      } else {
        selectedKey = null;
        legalTargets = [];
      }
      renderBoard();
    }

    function renderSideWindow() {
      // Turn display
      const turnEl = document.getElementById("turnDisplay");
      const activeSeat = SEATS[currentTurn];
      turnEl.innerHTML = `<span class="dot" style="background:${activeSeat.color};"></span><span>${activeSeat.name}'s Turn (${controllers[currentTurn].toUpperCase()})</span>`;

      // Players & Scores
      const grid = document.getElementById("playersGrid");
      grid.innerHTML = "";
      SEATS.forEach((s, idx) => {
        const card = document.createElement("div");
        card.className = `player-card ${currentTurn === idx && !gameOver ? "active" : ""} ${eliminated[idx] ? "eliminated" : ""}`;
        card.innerHTML = `
          <div class="player-header">
            <span style="color:${s.color};">${s.name}</span>
            <span>${scores[idx]} pts</span>
          </div>
          <select data-seat="${idx}">
            <option value="human" ${controllers[idx] === "human" ? "selected" : ""}>Human</option>
            <option value="ai" ${controllers[idx] === "ai" ? "selected" : ""}>AI Bot</option>
          </select>
          <div class="captured-strip" title="Captured pieces">${capturedBy[idx].join("") || "—"}</div>
        `;
        card.querySelector("select").addEventListener("change", (e) => {
          controllers[idx] = e.target.value;
          scheduleAiIfNeeded();
        });
        grid.appendChild(card);
      });

      // Move History Window
      const scroll = document.getElementById("movesScroll");
      document.getElementById("moveCounterBadge").textContent = `${moveHistory.length} Move${moveHistory.length === 1 ? "" : "s"}`;

      if (moveHistory.length === 0) {
        scroll.innerHTML = `<div class="empty-moves">No moves yet. Make your opening move on the board to see live move logs appear here!</div>`;
        return;
      }

      scroll.innerHTML = "";
      // Group moves into rounds of 3 seats
      const rounds = [];
      let currentRound = [null, null, null];
      moveHistory.forEach((m, index) => {
        if (currentRound[m.seat] !== null) {
          rounds.push(currentRound);
          currentRound = [null, null, null];
        }
        currentRound[m.seat] = { ...m, isLatest: index === moveHistory.length - 1 };
        if (m.seat === 2) {
          rounds.push(currentRound);
          currentRound = [null, null, null];
        }
      });
      if (currentRound.some(x => x !== null)) rounds.push(currentRound);

      rounds.forEach((round, rIdx) => {
        const row = document.createElement("div");
        row.className = "move-round";
        let html = `<div class="round-num">${rIdx + 1}.</div>`;
        for (let s = 0; s < 3; s++) {
          const item = round[s];
          if (!item) {
            html += `<div style="text-align:center; color:#475569;">…</div>`;
          } else {
            const cls = `move-pill ${item.isLatest ? "latest" : ""} ${item.isCapture ? "capture" : ""} ${item.isCheck ? "check" : ""}`;
            html += `<div class="${cls}"><span>${item.notation}</span>${item.gained ? `<span style="color:#fbbf24;font-size:0.68rem;">+${item.gained}</span>` : ""}</div>`;
          }
        }
        row.innerHTML = html;
        scroll.appendChild(row);
      });
      scroll.scrollTop = scroll.scrollHeight;
    }

    function renderAll() {
      renderBoard();
      renderSideWindow();
    }

    function startNewMatch() {
      clearTimeout(aiTimer);
      setupInitialPieces();
      currentTurn = 0;
      selectedKey = null;
      legalTargets = [];
      moveHistory = [];
      undoStack = [];
      scores = [0, 0, 0];
      capturedBy = [[], [], []];
      eliminated = [false, false, false];
      gameOver = false;
      lastMove = null;
      document.getElementById("alertText").textContent = "New match started! White moves first.";
      renderAll();
      scheduleAiIfNeeded();
    }

    document.getElementById("newGameBtn").addEventListener("click", startNewMatch);
    document.getElementById("undoBtn").addEventListener("click", () => {
      clearTimeout(aiTimer);
      if (undoStack.length > 0) {
        restoreState(undoStack.pop());
      }
    });
    document.getElementById("soundBtn").addEventListener("click", (e) => {
      soundOn = !soundOn;
      e.target.textContent = soundOn ? "🔊 Sound: On" : "🔇 Sound: Off";
    });
    document.getElementById("copyLogBtn").addEventListener("click", () => {
      const text = moveHistory.map((m, i) => `${i + 1}. [${SEATS[m.seat].name}] ${m.notation}`).join("\n");
      navigator.clipboard.writeText(text || "No moves yet.");
      document.getElementById("alertText").textContent = "📋 Move history copied to clipboard!";
    });
    document.getElementById("shareBtn").addEventListener("click", () => {
      const base = window.location.href.split("?")[0];
      navigator.clipboard.writeText(`${base}?seat=1`);
      document.getElementById("alertText").textContent = "🔗 Shareable seat link copied!";
    });

    startNewMatch();
  </script>

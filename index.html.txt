    function k(q, r) { return `${q},${r}`; }
    function parseK(str) { return str.split(',').map(Number); }
    function inBounds(q, r) {
      return Math.max(Math.abs(q), Math.abs(r), Math.abs(-q - r)) <= RADIUS;
    }
    function hexLabel(q, r) {

    function hexCenter(q, r) {

    function playTone(type) {
      if (!soundOn) return;
      try {
        const ctx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.connect(gain);
        gain.connect(ctx.destination);
        const now = ctx.currentTime;
        if (type === 'move') {
          osc.frequency.setValueAtTime(340, now);
          gain.gain.setValueAtTime(0.08, now);
          gain.gain.exponentialRampToValueAtTime(0.001, now + 0.09);
          osc.start(now); osc.stop(now + 0.09);
        } else if (type === 'capture') {
          osc.type = 'triangle';
          osc.frequency.setValueAtTime(220, now);
          osc.frequency.exponentialRampToValueAtTime(110, now + 0.14);
          gain.gain.setValueAtTime(0.14, now);
          gain.gain.exponentialRampToValueAtTime(0.001, now + 0.14);
          osc.start(now); osc.stop(now + 0.14);
        } else if (type === 'check') {
          osc.type = 'sawtooth';
          osc.frequency.setValueAtTime(520, now);
          osc.frequency.setValueAtTime(680, now + 0.08);
          gain.gain.setValueAtTime(0.12, now);
          gain.gain.exponentialRampToValueAtTime(0.001, now + 0.22);
          osc.start(now); osc.stop(now + 0.22);
        }
      } catch (e) {}
    }

    function getPseudoMoves(state, fromKey) {
      const piece = state[fromKey];
      if (!piece) return [];
      const [q, r] = parseK(fromKey);
      const moves = [];
      const addIfValid = (nq, nr) => {
        if (!inBounds(nq, nr)) return false;
        const tk = k(nq, nr);
        const occ = state[tk];
        if (!occ) {
          moves.push(tk);
          return true;
        }
        if (occ.owner !== piece.owner) {
          moves.push(tk);
        }
        return false;
      };

      if (piece.type === 'R' || piece.type === 'Q') {
        for (const [dq, dr] of ORTHO_DIRS) {
          let nq = q + dq, nr = r + dr;
          while (addIfValid(nq, nr)) { nq += dq; nr += dr; }
        }
      }
      if (piece.type === 'B' || piece.type === 'Q') {
        for (const [dq, dr] of DIAG_DIRS) {
          let nq = q + dq, nr = r + dr;
          while (addIfValid(nq, nr)) { nq += dq; nr += dr; }
        }
      }
      if (piece.type === 'N') {
        for (const [dq, dr] of KNIGHT_JUMPS) addIfValid(q + dq, r + dr);
      }
      if (piece.type === 'K') {
        for (const [dq, dr] of [...ORTHO_DIRS, ...DIAG_DIRS]) addIfValid(q + dq, r + dr);
      }
      if (piece.type === 'P') {
        const fwdDirs = piece.owner === 1
          ? [[0, -1], [1, -1]]
          : piece.owner === 2
          ? [[1, 0], [1, -1]]
          : [[-1, 1], [0, 1]];
        const capDirs = piece.owner === 1
          ? [[-1, 0], [1, -2], [2, -1], [-1, -1]]
          : piece.owner === 2
          ? [[0, 1], [2, -1], [1, -2], [0, -1]]
          : [[-1, 0], [1, 0], [-2, 1], [-1, 2]];

        for (const [dq, dr] of fwdDirs) {
          const nq = q + dq, nr = r + dr;
          if (inBounds(nq, nr) && !state[k(nq, nr)]) moves.push(k(nq, nr));
        }
        for (const [dq, dr] of capDirs) {
          const nq = q + dq, nr = r + dr;
          const tk = k(nq, nr);
          if (inBounds(nq, nr) && state[tk] && state[tk].owner !== piece.owner) {
            moves.push(tk);
          }
        }
      }
      return moves;
    }

    function isKingInCheck(state, seat) {
      let kingKey = null;
      for (const [sq, p] of Object.entries(state)) {
        if (p.owner === seat && p.type === 'K') { kingKey = sq; break; }
      }
      if (!kingKey) return true;
      for (const [sq, p] of Object.entries(state)) {
        if (p.owner !== seat && !eliminated[p.owner]) {
          const pm = getPseudoMoves(state, sq);
          if (pm.includes(kingKey)) return true;
        }
      }
      return false;
    }

    function getLegalMoves(state, fromKey) {
      const piece = state[fromKey];
      if (!piece) return [];
      const pseudo = getPseudoMoves(state, fromKey);
      return pseudo.filter(toKey => {
        const copy = JSON.parse(JSON.stringify(state));
        copy[toKey] = copy[fromKey];
        delete copy[fromKey];
        return !isKingInCheck(copy, piece.owner);
      });
    }

    function hasAnyLegalMove(state, seat) {
      for (const [sq, p] of Object.entries(state)) {
        if (p.owner === seat && getLegalMoves(state, sq).length > 0) return true;
      }
      return false;
    }

    function nextActiveSeat(curr) {
      for (let i = 1; i <= 3; i++) {
        const cand = ((curr - 1 + i) % 3) + 1;
        if (!eliminated[cand]) return cand;
      }
      return curr;
    }

    function executeMove(fromKey, toKey) {
      if (gameOver) return;
      const mover = board[fromKey];
      if (!mover || mover.owner !== turn) return;

      undoStack.push(JSON.stringify({
        board, turn, eliminated, scores, capturedBy, moveLog, lastMove, gameOver
      }));

      const target = board[toKey];
      let pts = 0;
      let capGlyph = '';
      if (target) {
        pts = VALUES[target.type] || 1;
        scores[mover.owner] += pts;
        capturedBy[mover.owner].push(GLYPHS[target.type]);
        capGlyph = '×' + GLYPHS[target.type];
        if (target.type === 'K') eliminated[target.owner] = true;
      }

      board[toKey] = mover;
      delete board[fromKey];

      // Auto-promote Pawn at far edge
      const [tq, tr] = parseK(toKey);
      if (mover.type === 'P' && (Math.abs(tq) === RADIUS || Math.abs(tr) === RADIUS || Math.abs(-tq - tr) === RADIUS)) {
        if ((mover.owner === 1 && tr === -RADIUS) ||
            (mover.owner === 2 && tq === RADIUS) ||
            (mover.owner === 3 && (tr === RADIUS || tq === -RADIUS))) {
          board[toKey].type = 'Q';
        }
      }

      lastMove = { from: fromKey, to: toKey };
      selectedKey = null;
      validTargets = [];

      // Check status on opponents
      let checkFlag = '';
      const ruleMode = document.getElementById('ruleMode').value;
      for (const s of [1, 2, 3]) {
        if (s !== mover.owner && !eliminated[s]) {
          if (isKingInCheck(board, s)) {
            checkFlag = '+';
            if (!hasAnyLegalMove(board, s)) {
              checkFlag = '#';
              eliminated[s] = true;
              scores[mover.owner] += 10;
              if (ruleMode === 'first_mate') {
                gameOver = true;
              }
            }
          }
        }
      }

      const [fq, fr] = parseK(fromKey);
      const notation = `${GLYPHS[mover.type]}${hexLabel(fq, fr)}→${hexLabel(tq, tr)}${capGlyph}${checkFlag}`;
      moveLog.push({
        seat: mover.owner,
        text: notation,
        isCap: !!target,
        isCheck: checkFlag !== '',
        pts
      });

      const aliveSeats = [1, 2, 3].filter(s => !eliminated[s]);
      if (aliveSeats.length <= 1) gameOver = true;

      if (checkFlag) playTone('check');
      else if (target) playTone('capture');
      else playTone('move');

      if (!gameOver) {
        turn = nextActiveSeat(turn);
      }

      renderAll();
      if (!gameOver) scheduleAiIfNeeded();
    }

    function scheduleAiIfNeeded() {
      clearTimeout(aiTimer);
      if (gameOver) return;
      const ctrl = document.getElementById(`ctrl${turn}`).value;
      if (ctrl === 'ai') {
        aiTimer = setTimeout(runAiTurn, 420);
      }
    }

    function runAiTurn() {
      if (gameOver) return;
      const allMoves = [];
      for (const [sq, p] of Object.entries(board)) {
        if (p.owner === turn) {
          const legals = getLegalMoves(board, sq);
          for (const dst of legals) {
            let weight = Math.random() * 2;
            const victim = board[dst];
            if (victim) weight += (VALUES[victim.type] || 1) * 10;
            const copy = JSON.parse(JSON.stringify(board));
            copy[dst] = copy[sq];
            delete copy[sq];
            for (const opp of [1, 2, 3]) {
              if (opp !== turn && !eliminated[opp] && isKingInCheck(copy, opp)) {
                weight += 14;
              }
            }
            allMoves.push({ from: sq, to: dst, weight });
          }
        }
      }
      if (allMoves.length === 0) {
        eliminated[turn] = true;
        const alive = [1, 2, 3].filter(s => !eliminated[s]);
        if (alive.length <= 1) gameOver = true;
        else turn = nextActiveSeat(turn);
        renderAll();
        scheduleAiIfNeeded();
        return;
      }
      allMoves.sort((a, b) => b.weight - a.weight);
      executeMove(allMoves[0].from, allMoves[0].to);
    }

    function renderBoard() {
      svg.innerHTML = '';
      const checkedSeats = {
        1: !eliminated[1] && isKingInCheck(board, 1),
        2: !eliminated[2] && isKingInCheck(board, 2),
        3: !eliminated[3] && isKingInCheck(board, 3)
      };

      for (let q = -RADIUS; q <= RADIUS; q++) {
        for (let r = -RADIUS; r <= RADIUS; r++) {
          if (!inBounds(q, r)) continue;
          const cellKey = k(q, r);
          const { x, y } = hexCenter(q, r);
          const shade = (((q - r) % 3) + 3) % 3;
          let fill = shade === 0 ? '#334155' : shade === 1 ? '#1e293b' : '#283548';

          if (lastMove && (lastMove.from === cellKey || lastMove.to === cellKey)) {
            fill = '#4b4524';
          }
          if (selectedKey === cellKey) {
            fill = '#0369a1';
          }

          const g = document.createElementNS('http://www.w3.org/2000/svg', 'g');
          g.setAttribute('class', 'hex-cell');

          const pts = [];
          for (let i = 0; i < 6; i++) {
            const ang = (Math.PI / 180) * (60 * i - 30);
            pts.push(`${x + (HEX_SIZE - 1.2) * Math.cos(ang)},${y + (HEX_SIZE - 1.2) * Math.sin(ang)}`);
          }
          const poly = document.createElementNS('http://www.w3.org/2000/svg', 'polygon');
          poly.setAttribute('points', pts.join(' '));
          poly.setAttribute('fill', fill);
          poly.setAttribute('stroke', '#0f172a');
          poly.setAttribute('stroke-width', '1.5');
          g.appendChild(poly);

          const lbl = document.createElementNS('http://www.w3.org/2000/svg', 'text');
          lbl.setAttribute('x', x);
          lbl.setAttribute('y', y + HEX_SIZE * 0.62);
          lbl.setAttribute('class', 'coord-label');
          lbl.textContent = hexLabel(q, r);
          g.appendChild(lbl);

          const piece = board[cellKey];
          if (piece) {
            if (piece.type === 'K' && checkedSeats[piece.owner]) {
              poly.setAttribute('fill', '#7f1d1d');
              poly.setAttribute('stroke', '#ef4444');
              poly.setAttribute('stroke-width', '2.5');
            }
            const txt = document.createElementNS('http://www.w3.org/2000/svg', 'text');
            txt.setAttribute('x', x);
            txt.setAttribute('y', y - 1);
            txt.setAttribute('class', 'piece-text');
            txt.setAttribute('fill', SEAT_COLORS[piece.owner]);
            txt.setAttribute('stroke', '#050811');
            txt.setAttribute('stroke-width', '0.8');
            txt.textContent = GLYPHS[piece.type];
            g.appendChild(txt);
          }

          if (validTargets.includes(cellKey)) {
            const dot = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
            dot.setAttribute('cx', x);
            dot.setAttribute('cy', y);
            dot.setAttribute('r', piece ? HEX_SIZE * 0.48 : 6.5);
            dot.setAttribute('fill', piece ? 'none' : 'rgba(34, 197, 94, 0.85)');
            if (piece) {
              dot.setAttribute('stroke', '#22c55e');
              dot.setAttribute('stroke-width', '3');
            }
            g.appendChild(dot);
          }

          g.addEventListener('click', () => onCellClick(cellKey));
          svg.appendChild(g);
        }
      }
    }

    function onCellClick(cellKey) {
      if (gameOver) return;
      if (document.getElementById(`ctrl${turn}`).value === 'ai') return;

      if (selectedKey && validTargets.includes(cellKey)) {
        executeMove(selectedKey, cellKey);
        return;
      }

      const piece = board[cellKey];
      if (piece && piece.owner === turn) {
        selectedKey = cellKey;
        validTargets = getLegalMoves(board, cellKey);
      } else {
        selectedKey = null;
        validTargets = [];
      }
      renderBoard();
    }

    function renderHud() {
      moveCounter.textContent = `${moveLog.length} move${moveLog.length === 1 ? '' : 's'}`;

      for (const s of [1, 2, 3]) {
        const card = document.getElementById(`seatCard${s}`);
        card.classList.toggle('active', !gameOver && turn === s);
        card.classList.toggle('eliminated', eliminated[s]);
        document.getElementById(`score${s}`).textContent = `${scores[s]} pts`;
        document.getElementById(`caps${s}`).textContent = capturedBy[s].join('');
      }

      if (gameOver) {
        const sorted = [1, 2, 3].sort((a, b) => scores[b] - scores[a]);
        const alive = [1, 2, 3].filter(s => !eliminated[s]);
        const winner = alive.length === 1 ? alive[0] : sorted[0];
        turnDot.style.background = SEAT_COLORS[winner];
        turnText.textContent = `🏆 Match Over — ${SEAT_NAMES[winner]} Wins (${scores[winner]} pts)!`;
      } else {
        turnDot.style.background = SEAT_COLORS[turn];
        const inChk = isKingInCheck(board, turn) ? ' • CHECK!' : '';
        turnText.textContent = `${SEAT_NAMES[turn]}'s Turn${inChk}`;
      }

      if (moveLog.length === 0) {
        movesFeed.innerHTML = `<div style="text-align:center;color:var(--muted);font-size:0.82rem;padding:34px 12px;">
          Click any piece on the board to make the opening move. All moves from White, Crimson &amp; Emerald appear here live.
        </div>`;
        return;
      }

      // Group moves into rounds of 3 seats
      const rounds = [];
      let currentRound = { 1: null, 2: null, 3: null };
      moveLog.forEach((m, idx) => {
        if (currentRound[m.seat] !== null) {
          rounds.push(currentRound);
          currentRound = { 1: null, 2: null, 3: null };
        }
        currentRound[m.seat] = { ...m, isLatest: idx === moveLog.length - 1 };
        if (m.seat === 3) {
          rounds.push(currentRound);
          currentRound = { 1: null, 2: null, 3: null };
        }
      });
      if (currentRound[1] || currentRound[2] || currentRound[3]) {
        rounds.push(currentRound);
      }

      movesFeed.innerHTML = rounds.map((rnd, rIdx) => {
        const cellHtml = (s) => {
          const item = rnd[s];
          if (!item) return `<div class="m-cell" style="opacity:0.25;">—</div>`;
          const cls = [
            'm-cell',
            item.isLatest ? 'latest' : '',
            item.isCap ? 'cap' : '',
            item.isCheck ? 'chk' : ''
          ].filter(Boolean).join(' ');
          return `<div class="${cls}"><span>${item.text}</span></div>`;
        };
        return `<div class="round-row">
          <div class="r-idx">${rIdx + 1}.</div>
          ${cellHtml(1)}
          ${cellHtml(2)}
          ${cellHtml(3)}
        </div>`;
      }).join('');

      movesFeed.scrollTop = movesFeed.scrollHeight;
    }

    function renderAll() {
      renderBoard();
      renderHud();
    }

      btn.textContent = 'Copied!';
      setTimeout(() => { btn.textContent = 'Copy Log'; }, 1400);
    });

    });

    setupBoard();

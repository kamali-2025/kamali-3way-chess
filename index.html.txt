    const RADIUS = 5;
    const HEX_SIZE = 32;
    const GLYPHS = { K: '♔', Q: '♕', R: '♖', B: '♗', N: '♘', P: '♙' };
    const VALS = { K: 15, Q: 9, R: 5, B: 3, N: 3, P: 1 };
    const NAMES = { 1: 'White', 2: 'Crimson', 3: 'Emerald' };
    const COLORS = { 1: '#f8fafc', 2: '#fb7185', 3: '#34d399' };

    const ORTHO = [
      [1, 0], [-1, 0], [0, 1], [0, -1], [1, -1], [-1, 1]
    ];
    const DIAG = [
      [2, -1], [1, 1], [-1, 2], [-2, 1], [-1, -1], [1, -2]
    ];
    const KNIGHT = [
      [3, -1], [3, -2], [2, 1], [1, 2], [-1, 3], [-2, 3],
      [-3, 2], [-3, 1], [-2, -1], [-1, -2], [1, -3], [2, -3]
    ];

    let board = {};
    let turn = 1;

    function playTone(type) {
      if (!soundOn) return;
      try {
        const ctx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.connect(gain);
        gain.connect(ctx.destination);
        const now = ctx.currentTime;
        if (type === 'move') {
          osc.frequency.setValueAtTime(340, now);
          osc.frequency.exponentialRampToValueAtTime(480, now + 0.08);
          gain.gain.setValueAtTime(0.08, now);
          gain.gain.exponentialRampToValueAtTime(0.001, now + 0.09);
        } else if (type === 'capture') {
          osc.type = 'triangle';
          osc.frequency.setValueAtTime(220, now);
          osc.frequency.exponentialRampToValueAtTime(110, now + 0.14);
          gain.gain.setValueAtTime(0.14, now);
          gain.gain.exponentialRampToValueAtTime(0.001, now + 0.15);
        } else {
          osc.type = 'sawtooth';
          osc.frequency.setValueAtTime(520, now);
          osc.frequency.setValueAtTime(680, now + 0.09);
          gain.gain.setValueAtTime(0.12, now);
          gain.gain.exponentialRampToValueAtTime(0.001, now + 0.22);
        }
        osc.start(now);
        osc.stop(now + 0.24);
      } catch (e) {}
    }

    function key(q, r) { return `${q},${r}`; }
    function parseKey(k) { return k.split(',').map(Number); }
    function inBounds(q, r) {
      return Math.max(Math.abs(q), Math.abs(r), Math.abs(-q - r)) <= RADIUS;
    }
    function hexLabel(q, r) {
      const col = String.fromCharCode(65 + q + RADIUS);
      const row = r + RADIUS + 1;
      return `${col}${row}`;
    }
    function hexToPixel(q, r) {
      return {
        x: HEX_SIZE * (Math.sqrt(3) * q + (Math.sqrt(3) / 2) * r),
        y: HEX_SIZE * (1.5 * r)
      };
    }

    function setupGame() {
      if (aiTimer) clearTimeout(aiTimer);
      board = {};
      turn = 1;
      eliminated = { 1: false, 2: false, 3: false };
      points = { 1: 0, 2: 0, 3: 0 };
      captured = { 1: [], 2: [], 3: [] };
      selected = null;

      moveHistory = [];
      snapshots = [];

      // Seat 1: White (Bottom)
      const p1Back = [
        [-4, 5, 'R'], [-3, 5, 'N'], [-2, 5, 'B'], [-1, 5, 'Q'],
        [0, 5, 'K'], [1, 5, 'B'], [2, 5, 'N'], [3, 5, 'R']
      ];
      p1Back.forEach(([q, r, t]) => board[key(q, r)] = { owner: 1, type: t });
      for (let q = -4; q <= 3; q++) board[key(q, 4)] = { owner: 1, type: 'P' };

      // Seat 2: Crimson (Top-Left)
      const p2Back = [
        [-5, 0, 'R'], [-5, 1, 'N'], [-5, 2, 'B'], [-5, 3, 'Q'],
        [-5, 4, 'K'], [-4, -1, 'B'], [-3, -2, 'N'], [-2, -3, 'R']
      ];
      p2Back.forEach(([q, r, t]) => board[key(q, r)] = { owner: 2, type: t });
      const p2Pawns = [[-4, 0], [-4, 1], [-4, 2], [-4, 3], [-3, -1], [-2, -2], [-1, -3], [0, -4]];
      p2Pawns.forEach(([q, r]) => board[key(q, r)] = { owner: 2, type: 'P' });

      // Seat 3: Emerald (Top-Right)
      const p3Back = [
        [5, -5, 'R'], [5, -4, 'N'], [5, -3, 'B'], [5, -2, 'Q'],

      ];
      p3Back.forEach(([q, r, t]) => board[key(q, r)] = { owner: 3, type: t });
      const p3Pawns = [[4, -4], [4, -3], [4, -2], [4, -1], [3, -4], [2, -4], [1, -4], [0, -5]];
      p3Pawns.forEach(([q, r]) => {

    function pawnDirs(owner) {
      if (owner === 1) return { step: [[0, -1], [1, -1]], cap: [[-1, 0], [1, -2], [2, -1], [-1, -1]] };
      if (owner === 2) return { step: [[1, 0], [1, -1]], cap: [[0, 1], [2, -1], [1, 1], [0, -1]] };
      return { step: [[-1, 1], [0, 1]], cap: [[-1, 0], [-1, 2], [-2, 1], [1, 0]] };
    }

    function pseudoMovesFrom(q, r, stateBoard) {
      const piece = stateBoard[key(q, r)];
      if (!piece) return [];
      const res = [];
      const addSlide = (dirs, maxSteps = 12) => {
        for (const [dq, dr] of dirs) {
          for (let s = 1; s <= maxSteps; s++) {
            const nq = q + dq * s, nr = r + dr * s;
            if (!inBounds(nq, nr)) break;
            const target = stateBoard[key(nq, nr)];
            if (!target) {
              res.push([nq, nr]);
            } else {
              if (target.owner !== piece.owner) res.push([nq, nr]);
              break;
            }
          }
        }
      };

      if (piece.type === 'R') addSlide(ORTHO);
      else if (piece.type === 'B') addSlide(DIAG);
      else if (piece.type === 'Q') { addSlide(ORTHO); addSlide(DIAG); }
      else if (piece.type === 'K') { addSlide(ORTHO, 1); addSlide(DIAG, 1); }
      else if (piece.type === 'N') {
        for (const [dq, dr] of KNIGHT) {
          const nq = q + dq, nr = r + dr;
          if (!inBounds(nq, nr)) continue;
          const target = stateBoard[key(nq, nr)];
          if (!target || target.owner !== piece.owner) res.push([nq, nr]);
        }
      } else if (piece.type === 'P') {
        const pd = pawnDirs(piece.owner);
        for (const [dq, dr] of pd.step) {
          const nq = q + dq, nr = r + dr;
          if (inBounds(nq, nr) && !stateBoard[key(nq, nr)]) res.push([nq, nr]);
        }
        for (const [dq, dr] of pd.cap) {
          const nq = q + dq, nr = r + dr;
          if (!inBounds(nq, nr)) continue;
          const target = stateBoard[key(nq, nr)];
          if (target && target.owner !== piece.owner) res.push([nq, nr]);
        }
      }
      return res;
    }

    function isKingInCheck(owner, stateBoard) {
      let kingPos = null;
      for (const [k, p] of Object.entries(stateBoard)) {
        if (p.owner === owner && p.type === 'K') { kingPos = k; break; }
      }
      if (!kingPos) return true;
      for (const [k, p] of Object.entries(stateBoard)) {
        if (p.owner === owner || eliminated[p.owner]) continue;
        const [q, r] = parseKey(k);
        const moves = pseudoMovesFrom(q, r, stateBoard);
        if (moves.some(([mq, mr]) => key(mq, mr) === kingPos)) return true;
      }
      return false;
    }

    function legalMovesFor(q, r) {
      const piece = board[key(q, r)];
      if (!piece) return [];
      return pseudoMovesFrom(q, r, board).filter(([nq, nr]) => {
        const clone = JSON.parse(JSON.stringify(board));
        clone[key(nq, nr)] = clone[key(q, r)];
        delete clone[key(q, r)];
        return !isKingInCheck(piece.owner, clone);
      });
    }

    function allLegalMoves(owner) {
      const list = [];
      for (const [k, p] of Object.entries(board)) {
        if (p.owner !== owner) continue;
        const [q, r] = parseKey(k);
        for (const [nq, nr] of legalMovesFor(q, r)) {
          list.push({ from: [q, r], to: [nq, nr] });
        }
      }
      return list;
    }

    function nextActiveTurn(curr) {
      for (let i = 1; i <= 3; i++) {
        const cand = ((curr - 1 + i) % 3) + 1;
        if (!eliminated[cand]) return cand;
      }
      return curr;
    }

    function executeMove(from, to) {
      if (gameOver) return;
      snapshots.push(JSON.stringify({
        board, turn, eliminated, points, captured, moveHistory, lastMove, gameOver
      }));

      const [fq, fr] = from;
      const [tq, tr] = to;
      const mover = board[key(fq, fr)];
      const victim = board[key(tq, tr)];

      let gained = 0;
      let capText = '';
      if (victim) {
        gained = VALS[victim.type] || 1;
        points[mover.owner] += gained;
        captured[mover.owner].push(GLYPHS[victim.type]);
        capText = '×' + GLYPHS[victim.type];
        if (victim.type === 'K') eliminated[victim.owner] = true;
      }

      board[key(tq, tr)] = mover;
      delete board[key(fq, fr)];

      // Pawn promotion
      if (mover.type === 'P' && (Math.abs(tq) === RADIUS || Math.abs(tr) === RADIUS || Math.abs(-tq - tr) === RADIUS)) {
        mover.type = 'Q';
      }

      lastMove = { from: key(fq, fr), to: key(tq, tr) };
      selected = null;
      validTargets = [];

      // Check opponents
      let suffix = '';
      let alertMsg = '';
      const mode = document.getElementById('ruleMode').value;

      for (const opp of [1, 2, 3]) {
        if (opp === mover.owner || eliminated[opp]) continue;
        const inChk = isKingInCheck(opp, board);
        const oppMoves = allLegalMoves(opp);
        if (inChk && oppMoves.length === 0) {
          suffix = '#';
          eliminated[opp] = true;
          points[mover.owner] += 10;
          alertMsg = `${NAMES[mover.owner]} checkmated ${NAMES[opp]}!`;
          if (mode === 'firstmate') {
            gameOver = true;
            alertMsg = `🏆 ${NAMES[mover.owner]} WINS by First Checkmate!`;
          }
        } else if (inChk) {
          suffix = '+';
          alertMsg = `⚠️ ${NAMES[opp]} is in CHECK!`;
        }
      }

      const alive = [1, 2, 3].filter(id => !eliminated[id]);
      if (alive.length <= 1) {
        gameOver = true;
        const winner = mode === 'points'
          ? [1, 2, 3].sort((a, b) => points[b] - points[a])[0]
          : alive[0];
        alertMsg = `🏆 ${NAMES[winner]} Wins the Arena!`;
      }

      moveHistory.push({
        owner: mover.owner,
        text: `${GLYPHS[mover.type]}${hexLabel(fq, fr)}→${hexLabel(tq, tr)}${capText}${suffix}`,
        isCap: !!victim,
        isChk: suffix === '+' || suffix === '#'
      });

      document.getElementById('alertText').textContent = alertMsg;
      playTone(suffix ? 'check' : (victim ? 'capture' : 'move'));

      if (!gameOver) {
        turn = nextActiveTurn(turn);
      }

      renderAll();
      scheduleAiIfNeeded();
    }

    function scheduleAiIfNeeded() {
      if (gameOver) return;
      const ctrl = document.getElementById('ctrl' + turn).value;
      if (ctrl === 'ai') {
        aiTimer = setTimeout(() => {
          const moves = allLegalMoves(turn);
          if (moves.length === 0) {
            eliminated[turn] = true;
            turn = nextActiveTurn(turn);
            renderAll();
            scheduleAiIfNeeded();
            return;
          }
          moves.sort((a, b) => {
            const va = board[key(a.to[0], a.to[1])];
            const vb = board[key(b.to[0], b.to[1])];
            return (vb ? VALS[vb.type] : 0) - (va ? VALS[va.type] : 0) + (Math.random() - 0.5);
          });
          executeMove(moves[0].from, moves[0].to);
        }, 420);
      }
    }

    function renderBoard() {
      svg.innerHTML = '';
      const palette = ['#334155', '#475569', '#1e293b'];
      for (let q = -RADIUS; q <= RADIUS; q++) {
        for (let r = -RADIUS; r <= RADIUS; r++) {
          if (!inBounds(q, r)) continue;
          const k = key(q, r);
          const { x, y } = hexToPixel(q, r);
          const g = document.createElementNS('http://www.w3.org/2000/svg', 'g');
          g.setAttribute('class', 'hex-cell');

          const pts = [];
          for (let i = 0; i < 6; i++) {
            const ang = (Math.PI / 180) * (60 * i - 30);
            pts.push(`${x + HEX_SIZE * Math.cos(ang)},${y + HEX_SIZE * Math.sin(ang)}`);
          }

          const poly = document.createElementNS('http://www.w3.org/2000/svg', 'polygon');
          poly.setAttribute('points', pts.join(' '));
          const cIdx = (((q - r) % 3) + 3) % 3;
          let fill = palette[cIdx];
          if (lastMove && (lastMove.from === k || lastMove.to === k)) fill = '#524b24';
          if (selected === k) fill = '#0369a1';
          poly.setAttribute('fill', fill);
          poly.setAttribute('stroke', '#0f172a');
          poly.setAttribute('stroke-width', '1.6');
          g.appendChild(poly);

          const lbl = document.createElementNS('http://www.w3.org/2000/svg', 'text');
          lbl.setAttribute('x', x);
          lbl.setAttribute('y', y + 18);
          lbl.setAttribute('class', 'coord-label');
          lbl.textContent = hexLabel(q, r);
          g.appendChild(lbl);

          const piece = board[k];
          if (piece) {
            if (piece.type === 'K' && isKingInCheck(piece.owner, board)) {
              const ring = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
              ring.setAttribute('cx', x);
              ring.setAttribute('cy', y);
              ring.setAttribute('r', 19);
              ring.setAttribute('fill', 'rgba(239, 68, 68, 0.35)');
              ring.setAttribute('stroke', '#ef4444');
              ring.setAttribute('stroke-width', '2');
              g.appendChild(ring);
            }
            const txt = document.createElementNS('http://www.w3.org/2000/svg', 'text');
            txt.setAttribute('x', x);
            txt.setAttribute('y', y - 1);
            txt.setAttribute('class', 'piece-text');
            txt.setAttribute('fill', COLORS[piece.owner]);
            txt.setAttribute('stroke', '#090d16');
            txt.setAttribute('stroke-width', '0.8');
            txt.textContent = GLYPHS[piece.type];
            g.appendChild(txt);
          }

          if (validTargets.some(([vq, vr]) => vq === q && vr === r)) {
            const dot = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
            dot.setAttribute('cx', x);
            dot.setAttribute('cy', y);
            dot.setAttribute('r', piece ? 17 : 7);
            dot.setAttribute('fill', piece ? 'none' : 'rgba(56, 189, 248, 0.78)');
            dot.setAttribute('stroke', '#38bdf8');
            dot.setAttribute('stroke-width', piece ? '2.5' : '0');
            g.appendChild(dot);
          }

          g.addEventListener('click', () => handleCellClick(q, r));
          svg.appendChild(g);
        }
      }
    }

    function handleCellClick(q, r) {
      if (gameOver) return;
      if (document.getElementById('ctrl' + turn).value === 'ai') return;
      const k = key(q, r);
      if (selected && validTargets.some(([vq, vr]) => vq === q && vr === r)) {
        executeMove(parseKey(selected), [q, r]);
        return;
      }
      const piece = board[k];
      if (piece && piece.owner === turn) {
        selected = k;
        validTargets = legalMovesFor(q, r);
      } else {
        selected = null;
        validTargets = [];
      }
      renderBoard();
    }

    function renderMoveWindow() {
      const turnEl = document.getElementById('turnDisplay');
      turnEl.innerHTML = gameOver
        ? `🏁 Match Finished`
        : `<span class="dot" style="background:${COLORS[turn]}"></span> ${NAMES[turn]}'s Turn`;

      for (const id of [1, 2, 3]) {
        const card = document.getElementById('card' + id);
        card.className = 'seat-mini' + (turn === id && !gameOver ? ' active' : '') + (eliminated[id] ? ' out' : '');
        document.getElementById('pts' + id).textContent = `${points[id]} pts`;
        document.getElementById('cap' + id).textContent = captured[id].join('');
      }

      document.getElementById('moveCountBadge').textContent = `${moveHistory.length} moves`;
      const listEl = document.getElementById('movesList');
      if (moveHistory.length === 0) {
        listEl.innerHTML = '<div class="empty-msg">Make a move on the board to see live moves appear here.</div>';
        return;
      }

      const rounds = [];
      let cur = { 1: null, 2: null, 3: null };
      moveHistory.forEach((m, idx) => {
        if (m.owner === 1 && (cur[1] || cur[2] || cur[3])) {
          rounds.push(cur);
          cur = { 1: null, 2: null, 3: null };
        }
        cur[m.owner] = { ...m, isLatest: idx === moveHistory.length - 1 };
      });
      rounds.push(cur);

      listEl.innerHTML = rounds.map((rnd, i) => {
        const cell = (seat) => {
          const item = rnd[seat];
          if (!item) return `<div></div>`;
          const cls = 'm-chip' + (item.isLatest ? ' latest' : '') + (item.isCap ? ' cap' : '') + (item.isChk ? ' chk' : '');
          return `<div class="${cls}">${item.text}</div>`;
        };
        return `<div class="round-row"><div class="r-num">${i + 1}</div>${cell(1)}${cell(2)}${cell(3)}</div>`;
      }).join('');

      listEl.scrollTop = listEl.scrollHeight;
    }

    function renderAll() {
      renderBoard();
      renderMoveWindow();
    }

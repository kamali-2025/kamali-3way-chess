  <script>
    const RADIUS = 5;
    const HEX_SIZE = 32;
    const GLYPHS = { K: '♔', Q: '♕', R: '♖', B: '♗', N: '♘', P: '♙' };
    const VALUES = { K: 15, Q: 9, R: 5, B: 3, N: 3, P: 1 };
    const NAMES = { 1: 'White', 2: 'Crimson', 3: 'Emerald' };
    const COLORS = { 1: '#f8fafc', 2: '#fb7185', 3: '#34d399' };

    let board = {};
    let turn = 1;
    let eliminated = { 1: false, 2: false, 3: false };
    let points = { 1: 0, 2: 0, 3: 0 };
    let captured = { 1: [], 2: [], 3: [] };
    let selected = null;
    let legalTargets = [];
    let moveHistory = [];
    let snapshots = [];
    let lastMove = null;
    let gameOver = false;
    let soundOn = true;
    let botTimer = null;

    const svg = document.getElementById('hexBoard');
    const movesListEl = document.getElementById('movesList');
    const moveCounterEl = document.getElementById('moveCounter');
    const statusTextEl = document.getElementById('statusText');
    const turnDotEl = document.getElementById('turnDot');
    const boardArenaEl = document.getElementById('boardArena');

    function key(q, r) { return `${q},${r}`; }
    function parseKey(k) { return k.split(',').map(Number); }
    function validHex(q, r) {
      return Math.max(Math.abs(q), Math.abs(r), Math.abs(-q - r)) <= RADIUS;
    }
    function hexToPixel(q, r) {
      return {
        x: HEX_SIZE * (Math.sqrt(3) * q + (Math.sqrt(3) / 2) * r),
        y: HEX_SIZE * (1.5 * r)
      };
    }
    function coordLabel(q, r) {
      const file = String.fromCharCode(97 + (q + 5));
      const rank = 6 - r;
      return `${file}${rank}`;
    }

    function playTone(freq, dur = 0.08, type = 'triangle') {
      if (!soundOn) return;
      try {
        const ctx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type = type;
        osc.frequency.value = freq;
        gain.gain.setValueAtTime(0.08, ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + dur);
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.start();
        osc.stop(ctx.currentTime + dur);
      } catch (e) {}
    }

    function cloneState() {
      return JSON.parse(JSON.stringify({
        board, turn, eliminated, points, captured, moveHistory, lastMove, gameOver
      }));
    }

    function initGame() {
      clearTimeout(botTimer);
      board = {};
      turn = 1;
      eliminated = { 1: false, 2: false, 3: false };
      points = { 1: 0, 2: 0, 3: 0 };
      captured = { 1: [], 2: [], 3: [] };
      selected = null;
      legalTargets = [];
      moveHistory = [];
      snapshots = [];
      lastMove = null;
      gameOver = false;

      // Seat 1 (White - Bottom)
      const p1Back = [
        [-4, 5, 'R'], [-3, 5, 'N'], [-2, 5, 'B'], [-1, 5, 'Q'], [0, 5, 'K'], [1, 4, 'B'], [2, 3, 'N'], [3, 2, 'R']
      ];
      p1Back.forEach(([q, r, t]) => board[key(q, r)] = { owner: 1, type: t });
      for (let q = -4; q <= 3; q++) {
        const r = q <= 0 ? 4 : 4 - q;
        if (validHex(q, r)) board[key(q, r)] = { owner: 1, type: 'P' };
      }

      // Seat 2 (Crimson - Top-Left)
      const p2Back = [
        [-5, 0, 'R'], [-5, 1, 'N'], [-5, 2, 'B'], [-5, 3, 'Q'], [-5, 4, 'K'], [-4, 4, 'B'], [-3, 4, 'N'], [-2, 4, 'R']
      ];
      p2Back.forEach(([q, r, t]) => { if (!board[key(q, r)]) board[key(q, r)] = { owner: 2, type: t }; });
      [[-4, 0], [-4, 1], [-4, 2], [-4, 3], [-3, 3], [-2, 3], [-1, 3]].forEach(([q, r]) => {
        if (!board[key(q, r)]) board[key(q, r)] = { owner: 2, type: 'P' };
      });

      // Seat 3 (Emerald - Top-Right)
      const p3Back = [
        [5, -5, 'R'], [5, -4, 'N'], [5, -3, 'B'], [5, -2, 'Q'], [5, -1, 'K'], [4, 0, 'B'], [3, 1, 'N'], [2, 2, 'R']
      ];
      p3Back.forEach(([q, r, t]) => { if (!board[key(q, r)]) board[key(q, r)] = { owner: 3, type: t }; });
      [[4, -4], [4, -3], [4, -2], [4, -1], [3, 0], [2, 1], [1, 2]].forEach(([q, r]) => {
        if (!board[key(q, r)]) board[key(q, r)] = { owner: 3, type: 'P' };
      });

      renderAll();
      scheduleBotIfNeeded();
    }

    const ORTHO_DIRS = [[1,0],[-1,0],[0,1],[0,-1],[1,-1],[-1,1]];
    const DIAG_DIRS = [[2,-1],[1,1],[-1,2],[-2,1],[-1,-1],[1,-2]];
    const KNIGHT_JUMPS = [
      [2,1],[1,2],[-1,3],[-2,3],[-3,2],[-3,1],
      [-2,-1],[-1,-2],[1,-3],[2,-3],[3,-2],[3,-1]
    ];

    function getPseudoMoves(fromKey, stBoard = board) {
      const piece = stBoard[fromKey];
      if (!piece) return [];
      const [q, r] = parseKey(fromKey);
      const moves = [];
      const addIfValid = (nq, nr) => {
        if (!validHex(nq, nr)) return false;
        const tk = key(nq, nr);
        const target = stBoard[tk];
        if (!target) {
          moves.push(tk);
          return true;
        }
        if (target.owner !== piece.owner) {
          moves.push(tk);
        }
        return false;
      };

      if (piece.type === 'R' || piece.type === 'Q') {
        ORTHO_DIRS.forEach(([dq, dr]) => {
          for (let i = 1; i <= 10; i++) if (!addIfValid(q + dq * i, r + dr * i)) break;
        });
      }
      if (piece.type === 'B' || piece.type === 'Q') {
        DIAG_DIRS.forEach(([dq, dr]) => {
          for (let i = 1; i <= 10; i++) if (!addIfValid(q + dq * i, r + dr * i)) break;
        });
      }
      if (piece.type === 'K') {
        [...ORTHO_DIRS, ...DIAG_DIRS].forEach(([dq, dr]) => addIfValid(q + dq, r + dr));
      }
      if (piece.type === 'N') {
        KNIGHT_JUMPS.forEach(([dq, dr]) => addIfValid(q + dq, r + dr));
      }
      if (piece.type === 'P') {
        const fwd = piece.owner === 1 ? [[0,-1],[1,-1]] : piece.owner === 2 ? [[1,0],[1,-1]] : [[-1,0],[-1,1]];
        const cap = piece.owner === 1 ? [[-1,0],[1,-2]] : piece.owner === 2 ? [[0,1],[2,-1]] : [[0,-1],[-2,1]];
        fwd.forEach(([dq, dr]) => {
          const nq = q + dq, nr = r + dr, tk = key(nq, nr);
          if (validHex(nq, nr) && !stBoard[tk]) moves.push(tk);
        });
        cap.forEach(([dq, dr]) => {
          const nq = q + dq, nr = r + dr, tk = key(nq, nr);
          if (validHex(nq, nr) && stBoard[tk] && stBoard[tk].owner !== piece.owner) moves.push(tk);
        });
      }
      return moves;
    }

    function isKingInCheck(owner, stBoard = board) {
      let kingKey = null;
      for (const [k, p] of Object.entries(stBoard)) {
        if (p.owner === owner && p.type === 'K') { kingKey = k; break; }
      }
      if (!kingKey) return true;
      for (const [k, p] of Object.entries(stBoard)) {
        if (p.owner !== owner && !eliminated[p.owner]) {
          if (getPseudoMoves(k, stBoard).includes(kingKey)) return true;
        }
      }
      return false;
    }

    function getLegalMoves(fromKey) {
      const piece = board[fromKey];
      if (!piece) return [];
      return getPseudoMoves(fromKey, board).filter(toKey => {
        const copy = { ...board };
        copy[toKey] = copy[fromKey];
        delete copy[fromKey];
        return !isKingInCheck(piece.owner, copy);
      });
    }

    function hasAnyLegalMove(owner) {
      for (const [k, p] of Object.entries(board)) {
        if (p.owner === owner && getLegalMoves(k).length > 0) return true;
      }
      return false;
    }

    function executeMove(fromKey, toKey) {
      if (gameOver) return;
      snapshots.push(cloneState());
      const mover = board[fromKey];
      const target = board[toKey];
      let gained = 0;

      if (target) {
        gained = VALUES[target.type] || 1;
        points[mover.owner] += gained;
        captured[mover.owner].push(GLYPHS[target.type]);
        if (target.type === 'K') eliminated[target.owner] = true;
      }

      board[toKey] = mover;
      delete board[fromKey];

      // Pawn promotion on far edge
      const [tq, tr] = parseKey(toKey);
      if (mover.type === 'P' && (Math.abs(tq) === RADIUS || Math.abs(tr) === RADIUS || Math.abs(-tq - tr) === RADIUS)) {
        board[toKey] = { owner: mover.owner, type: 'Q' };
      }

      lastMove = { from: fromKey, to: toKey };
      selected = null;
      legalTargets = [];

      // Check status on opponents
      let gaveCheck = false;
      let gaveMate = false;
      [1, 2, 3].forEach(opp => {
        if (opp !== mover.owner && !eliminated[opp]) {
          if (isKingInCheck(opp)) {
            gaveCheck = true;
            if (!hasAnyLegalMove(opp)) {
              gaveMate = true;
              eliminated[opp] = true;
              points[mover.owner] += 15;
            }
          }
        }
      });

      const notation = `${GLYPHS[mover.type]}${coordLabel(...parseKey(fromKey))}${target ? '×' : '→'}${coordLabel(tq, tr)}${gaveMate ? '#' : gaveCheck ? '+' : ''}`;
      moveHistory.push({
        owner: mover.owner,
        text: notation,
        capture: !!target,
        check: gaveCheck || gaveMate,
        gained
      });

      if (gaveMate || gaveCheck) playTone(560, 0.16, 'sawtooth');
      else if (target) playTone(380, 0.1, 'square');
      else playTone(260, 0.06, 'triangle');

      const ruleMode = document.getElementById('ruleMode').value;
      const alive = [1, 2, 3].filter(id => !eliminated[id]);

      if (ruleMode === 'first_mate' && gaveMate) {
        gameOver = true;
        statusTextEl.textContent = `🏆 ${NAMES[mover.owner]} wins by First Checkmate!`;
        renderAll();
        return;
      }
      if (alive.length <= 1) {
        gameOver = true;
        const winner = alive[0] || mover.owner;
        statusTextEl.textContent = `🏆 ${NAMES[winner]} wins the Arena (${points[winner]} pts)!`;
        renderAll();
        return;
      }

      // Advance turn
      for (let i = 1; i <= 3; i++) {
        const cand = ((turn - 1 + i) % 3) + 1;
        if (!eliminated[cand]) {
          turn = cand;
          break;
        }
      }

      renderAll();
      scheduleBotIfNeeded();
    }

    function scheduleBotIfNeeded() {
      clearTimeout(botTimer);
      if (gameOver) return;
      const ctrl = document.getElementById(`ctrl${turn}`).value;
      if (ctrl === 'bot') {
        botTimer = setTimeout(runBotMove, 420);
      }
    }

    function runBotMove() {
      if (gameOver) return;
      const candidates = [];
      for (const [k, p] of Object.entries(board)) {
        if (p.owner === turn) {
          getLegalMoves(k).forEach(to => {
            let score = Math.random() * 1.5;
            if (board[to]) score += (VALUES[board[to].type] || 1) * 10;
            const copy = { ...board, [to]: p };
            delete copy[k];
            [1, 2, 3].forEach(opp => {
              if (opp !== turn && !eliminated[opp] && isKingInCheck(opp, copy)) score += 8;
            });
            candidates.push({ from: k, to, score });
          });
        }
      }
      if (candidates.length === 0) {
        eliminated[turn] = true;
        for (let i = 1; i <= 3; i++) {
          const cand = ((turn - 1 + i) % 3) + 1;
          if (!eliminated[cand]) { turn = cand; break; }
        }
        renderAll();
        scheduleBotIfNeeded();
        return;
      }
      candidates.sort((a, b) => b.score - a.score);
      executeMove(candidates[0].from, candidates[0].to);
    }

    function renderBoard() {
      svg.innerHTML = '';
      const checkedKings = new Set();
      [1, 2, 3].forEach(id => {
        if (!eliminated[id] && isKingInCheck(id)) {
          for (const [k, p] of Object.entries(board)) {
            if (p.owner === id && p.type === 'K') checkedKings.add(k);
          }
        }
      });

      for (let q = -RADIUS; q <= RADIUS; q++) {
        for (let r = -RADIUS; r <= RADIUS; r++) {
          if (!validHex(q, r)) continue;
          const k = key(q, r);
          const { x, y } = hexToPixel(q, r);
          const pts = [];
          for (let i = 0; i < 6; i++) {
            const ang = (Math.PI / 180) * (60 * i - 30);
            pts.push(`${(x + (HEX_SIZE - 1) * Math.cos(ang)).toFixed(1)},${(y + (HEX_SIZE - 1) * Math.sin(ang)).toFixed(1)}`);
          }

          const shade = (((q - r) % 3) + 3) % 3;
          let fill = shade === 0 ? '#334155' : shade === 1 ? '#475569' : '#1e293b';
          if (lastMove && (lastMove.from === k || lastMove.to === k)) fill = '#5b5225';
          if (selected === k) fill = '#0284c7';
          if (checkedKings.has(k)) fill = '#b91c1c';

          const g = document.createElementNS('http://www.w3.org/2000/svg', 'g');
          g.setAttribute('class', 'hex-cell');

          const poly = document.createElementNS('http://www.w3.org/2000/svg', 'polygon');
          poly.setAttribute('points', pts.join(' '));
          poly.setAttribute('fill', fill);
          poly.setAttribute('stroke', '#0f172a');
          poly.setAttribute('stroke-width', '1.5');
          g.appendChild(poly);

          const lbl = document.createElementNS('http://www.w3.org/2000/svg', 'text');
          lbl.setAttribute('x', x);
          lbl.setAttribute('y', y + 18);
          lbl.setAttribute('class', 'coord-label');
          lbl.textContent = coordLabel(q, r);
          g.appendChild(lbl);

          if (legalTargets.includes(k)) {
            const dot = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
            dot.setAttribute('cx', x);
            dot.setAttribute('cy', y);
            dot.setAttribute('r', board[k] ? '15' : '6.5');
            dot.setAttribute('fill', board[k] ? 'none' : 'rgba(34, 197, 94, 0.85)');
            dot.setAttribute('stroke', '#22c55e');
            dot.setAttribute('stroke-width', '2.5');
            g.appendChild(dot);
          }

          const piece = board[k];
          if (piece) {
            const txt = document.createElementNS('http://www.w3.org/2000/svg', 'text');
            txt.setAttribute('x', x);
            txt.setAttribute('y', y - 1);
            txt.setAttribute('class', 'piece-text');
            txt.setAttribute('fill', COLORS[piece.owner]);
            txt.setAttribute('stroke', '#090d16');
            txt.setAttribute('stroke-width', '0.8');
            txt.textContent = GLYPHS[piece.type];
            g.appendChild(txt);
          }

          g.addEventListener('click', () => handleCellClick(k));
          svg.appendChild(g);
        }
      }
    }

    function handleCellClick(k) {
      if (gameOver) return;
      if (document.getElementById(`ctrl${turn}`).value === 'bot') return;

      if (selected && legalTargets.includes(k)) {
        executeMove(selected, k);
        return;
      }
      const piece = board[k];
      if (piece && piece.owner === turn) {
        selected = k;
        legalTargets = getLegalMoves(k);
        renderBoard();
      } else {
        selected = null;
        legalTargets = [];
        renderBoard();
      }
    }

    function renderHud() {
      [1, 2, 3].forEach(id => {
        const card = document.getElementById(`seatCard${id}`);
        card.classList.toggle('active', turn === id && !gameOver);
        card.classList.toggle('eliminated', eliminated[id]);
        document.getElementById(`pts${id}`).textContent = `${points[id]} pts`;
        document.getElementById(`cap${id}`).textContent = captured[id].join('');
      });

      if (!gameOver) {
        turnDotEl.style.background = COLORS[turn];
        const inChk = isKingInCheck(turn);
        statusTextEl.textContent = `${NAMES[turn]}'s Turn${inChk ? ' — CHECK!' : ''}`;
      }

      moveCounterEl.textContent = `${moveHistory.length} move${moveHistory.length === 1 ? '' : 's'}`;

      if (moveHistory.length === 0) {
        movesListEl.innerHTML = '<div class="empty-msg">Make a move on the board to see live moves appear right here.</div>';
        return;
      }

      // Group moves into rounds of 3 turns
      const rounds = [];
      let cur = { 1: null, 2: null, 3: null };
      moveHistory.forEach((m, idx) => {
        if (cur[m.owner] !== null || (m.owner === 1 && (cur[2] || cur[3]))) {
          rounds.push(cur);
          cur = { 1: null, 2: null, 3: null };
        }
        cur[m.owner] = { ...m, isLatest: idx === moveHistory.length - 1 };
      });
      rounds.push(cur);

      movesListEl.innerHTML = rounds.map((r, rIdx) => {
        const cell = (seat) => {
          const m = r[seat];
          if (!m) return '<div></div>';
          const cls = ['m-chip', m.isLatest ? 'latest' : '', m.capture ? 'capture' : '', m.check ? 'check' : ''].filter(Boolean).join(' ');
          return `<div class="${cls}"><span>${m.text}</span>${m.gained ? `<span style="color:#facc15;font-size:0.68rem;">+${m.gained}</span>` : ''}</div>`;
        };
        return `<div class="round-row"><div class="r-num">${rIdx + 1}.</div>${cell(1)}${cell(2)}${cell(3)}</div>`;
      }).join('');

      movesListEl.scrollTop = movesListEl.scrollHeight;
    }

    function renderAll() {
      renderBoard();
      renderHud();
    }

    document.getElementById('newBtn').addEventListener('click', initGame);
    document.getElementById('undoBtn').addEventListener('click', () => {
      if (snapshots.length === 0) return;
      clearTimeout(botTimer);
      const prev = snapshots.pop();
      board = prev.board;
      turn = prev.turn;
      eliminated = prev.eliminated;
      points = prev.points;
      captured = prev.captured;
      moveHistory = prev.moveHistory;
      lastMove = prev.lastMove;
      gameOver = prev.gameOver;
      selected = null;
      legalTargets = [];
      renderAll();
    });
    document.getElementById('soundBtn').addEventListener('click', (e) => {
      soundOn = !soundOn;
      e.target.textContent = `Sound: ${soundOn ? 'ON' : 'OFF'}`;
    });
    document.getElementById('toggleHudBtn').addEventListener('click', (e) => {
      const collapsed = boardArenaEl.classList.toggle('hud-collapsed');
      e.target.textContent = collapsed ? 'Show Moves HUD' : 'Hide Moves HUD';
    });
    document.getElementById('copyLogBtn').addEventListener('click', () => {
      const txt = moveHistory.map((m, i) => `${i + 1}. [${NAMES[m.owner]}] ${m.text}`).join('\n');
      navigator.clipboard?.writeText(txt || 'No moves yet.');
    });
    [1, 2, 3].forEach(id => {
      document.getElementById(`ctrl${id}`).addEventListener('change', scheduleBotIfNeeded);
    });

    initGame();
  </script>
